从日志上看，最终是走到了doBilling 的最后一个else；那么我们想办法把`MessageUtil.getInstance().free`的值搞成`1`是不是就ok了呢，

#### 5\. 编写frida脚本验证猜想
```
// lesson02.js
    function main() {
        Java.perform(function () {
            var MessageUtilHandler = Java.use('com.mydefinemmpay.tool.MessageUtil')
    
            if (MessageUtilHandler != undefined) {
    			//静态函数主动调用
                var instance = MessageUtilHandler.getInstance();
                console.log("===== " + JSON.stringify(instance))
                //成员变量主动修改
    			instance.free.value = "1"
            }
    
        })
    
    }
    
    setTimeout(main)
```

##### 6\. 执行`frida -UF hfdcxy.com.myapplication -l lesson02.js `
